Codeforces Problem 467A - George and Accommodation

import java.util.Scanner;

public class GeorgeandAcco{
    public static void main(String[] args) {
       Scanner sc = new Scanner(System.in);
       int n = sc.nextInt();
       sc.nextLine();
       int live,cap;
       int avail=0;
       for (int i = 0; i < n; i++) {
        live = sc.nextInt();
        cap = sc.nextInt();
        if (cap-live>=2) {
            avail++;
        } 
       }
       System.out.println(avail);
       sc.close();
    }
}
