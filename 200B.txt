Codeforces Problem 200B - Drinks

import java.util.Scanner;

public class Drinks {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        float total = 0;
        for (int i = 0; i < n; i++) {
            int ith = sc.nextInt();
            total+=ith;
        }
        System.out.println(total/n);
        sc.close();
    }
}
